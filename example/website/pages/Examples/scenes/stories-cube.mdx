# Stories Cube

Full-screen stories that turn to the next person like the faces of a cube.

<SceneLayout slug="stories-cube" bar="light">

### How it works

1. `itemSize` is the width of the 9:16 page, and `itemAnimation` just lines the pages up side by side and hides any page at ±1 or beyond.
2. Inside each page, `relativeProgress` sets `rotateY` to value × 90° with `perspective` 900 per 330 pt of width, turning on the right edge when leaving and on the left edge when entering, so the faces meet like a cube.
3. `onSnapToItem` restarts the bar over the 4000 ms `autoplayInterval`, `onScrollStart` freezes it, and a tap on the left third or the rest of the card calls `ref.prev()` or `ref.next()`.

### Uses

- [`itemAnimation`](/custom-animations)
- [`itemSize`](/props)

function StoriesCube() {
	const { width, height, insets } = useScene();
	const ref = React.useRef<CarouselRef>(null);
	const active = useSharedValue(0);
	const fill = useSharedValue(0);

	// A 9:16 story card above the reply bar, as wide as the screen allows.
	const room = height - insets.top - Math.max(insets.bottom, 8) - FOOTER;
	const cardHeight = Math.min(Math.round((width * 16) / 9), room);
	const card = { width: Math.round((cardHeight * 9) / 16), height: cardHeight };

	const startBar = React.useCallback(
		(index: number) => {
			active.value = index;
			fill.value = 0;
			fill.value = withTiming(1, { duration: INTERVAL, easing: Easing.linear });
		},
		[active, fill],
	);
	React.useEffect(() => startBar(0), [startBar]);

	const itemAnimation = React.useCallback<CarouselItemAnimation>(
		(value) => {
			"worklet";
			return {
				transform: [
					{ translateX: value * card.width + (width - card.width) / 2 },
				],
				opacity: Math.abs(value) >= 1 ? 0 : 1,
			};
		},
		[card.width, width],
	);

	// Tap the left third for the previous person, anywhere else on the card for the next.
	const tap = React.useMemo(
		() =>
			Gesture.Tap()
				.runOnJS(true)
				.onEnd((e, success) => {
					if (!success || e.y > card.height) return;
					if (e.x < width / 3) ref.current?.prev();
					else ref.current?.next();
				}),
		[card.height, width],
	);

	const renderItem = React.useCallback(
		({
			item,
			index,
			relativeProgress,
		}: {
			item: Story;
			index: number;
			relativeProgress: SharedValue<number>;
		}) => (
			<StoryPage
				story={item}
				index={index}
				card={card}
				relativeProgress={relativeProgress}
				active={active}
				fill={fill}
			/>
		),
		[card.width, card.height, active, fill],
	);

	return (
		<View
			style={[
				styles.screen,
				{ paddingTop: insets.top, paddingBottom: Math.max(insets.bottom, 8) },
			]}
		>
			<GestureDetector gesture={tap}>
				<View collapsable={false}>
					<Carousel
						ref={ref}
						data={STORIES}
						loop
						autoplay
						autoplayInterval={INTERVAL}
						itemSize={card.width}
						style={{ width, height: card.height + FOOTER }}
						animation={{
							type: "timing",
							duration: 700,
							easing: EASE_OUT_QUART,
						}}
						itemAnimation={itemAnimation}
						onScrollStart={() => cancelAnimation(fill)}
						onSnapToItem={startBar}
						renderItem={renderItem}
					/>
				</View>
			</GestureDetector>
		</View>
	);
}
