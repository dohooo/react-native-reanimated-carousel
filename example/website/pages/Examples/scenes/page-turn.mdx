# Page Turn

A small book of Tang poems whose pages turn around the spine, with the art on the back of each leaf.

<SceneLayout slug="page-turn" bar="light">

### How it works

1. `itemSize` is one page wide and `itemAnimation` hangs all 8 leaves on the spine, with the leaf in the air (value between −1 and 0) on top at `zIndex` 300.
2. Inside each leaf, `relativeProgress` turns the front from 0° to −180° with `perspective` 1400 around its left edge, and the back follows 180° behind with `backfaceVisibility` hidden.
3. With `loop` false, `autoplay` turns a page every 2400 ms; when `onSnapToItem` reports the last page, `ref.scrollTo({ index: 0 })` closes the book and autoplay starts again.

interface LeafProps {
	leaf: number;
	page: { width: number; height: number; top: number };
	relativeProgress: SharedValue<number>;
}

// The front turns from 0° to −180° around the spine; the back follows 180° behind it.
const Leaf = React.memo(function Leaf({
	leaf,
	page,
	relativeProgress,
}: LeafProps) {
	const scale = page.width / PAGE.width;
	const angle = useDerivedValue(() =>
		interpolate(
			relativeProgress.value,
			[-1, 0],
			[-180, 0],
			Extrapolation.CLAMP,
		),
	);
	const front = useAnimatedStyle(() => ({
		transform: [
			{ perspective: 1400 * scale },
			{ rotateY: `${angle.value}deg` },
		],
	}));
	const back = useAnimatedStyle(() => ({
		transform: [
			{ perspective: 1400 * scale },
			{ rotateY: `${angle.value + 180}deg` },
		],
	}));
	const shade = useAnimatedStyle(() => ({
		opacity: interpolate(
			relativeProgress.value,
			[-1, -0.5, 0],
			[0, 0.45, 0],
			Extrapolation.CLAMP,
		),
	}));
	const fit = [styles.sheet, { transform: [{ scale }] }];
	return (
		<>
			<Animated.View style={[styles.side, styles.frontSide, page, front]}>
				<View style={fit}>
					<Front leaf={leaf} />
					<Shade style={shade} />
				</View>
			</Animated.View>
			<Animated.View
				style={[
					styles.side,
					styles.backSide,
					page,
					{ left: -page.width },
					back,
				]}
			>
				<View style={fit}>
					<Back leaf={leaf} />
					<Shade style={shade} spineRight />
				</View>
			</Animated.View>
		</>
	);
});

function PageTurn() {
	const { width, height, insets } = useScene();
	const ref = React.useRef<CarouselRef>(null);
	const progress = useSharedValue(0);
	const [settled, setSettled] = React.useState(0); // the page the book came to rest on
	const [shown, setShown] = React.useState(0); // the nearest page, for the caption

	useAnimatedReaction(
		() => Math.min(LAST, Math.max(0, Math.round(progress.value))),
		(current, previous) => {
			if (current !== previous) scheduleOnRN(setShown, current);
		},
	);

	// Two pages meet at the spine in the middle of the screen.
	const room = height - insets.top - Math.max(insets.bottom, 12) - 280;
	const pageWidth = Math.floor(
		Math.min((width - 26) / 2, (room * PAGE.width) / PAGE.height),
	);
	const pageHeight = Math.round((pageWidth * PAGE.height) / PAGE.width);
	const lift = Math.round(pageHeight * 0.1); // headroom for a page that rises toward you as it turns
	const spine = width / 2;
	const page = { width: pageWidth, height: pageHeight, top: lift };
	const stage = { width, height: pageHeight + 2 * lift };

	const itemAnimation = React.useCallback<CarouselItemAnimation>(
		(value) => {
			"worklet";
			// Every leaf hangs on the spine and the one in the air stays on top.
			// Only the two facing pages and the turning leaf are drawn.
			const turning = value > -1 && value < 0;
			return {
				transform: [{ translateX: spine }],
				zIndex: turning
					? 300
					: value <= -1
						? 200 + Math.round(value)
						: 100 - Math.round(value),
				opacity: value > -2 && value < 1 ? 1 : 0,
			};
		},
		[spine],
	);

	// After the last page, close the book and start again.
	React.useEffect(() => {
		if (settled !== LAST) return;
		const timer = setTimeout(
			() => ref.current?.scrollTo({ index: 0 }),
			INTERVAL,
		);
		return () => clearTimeout(timer);
	}, [settled]);

	// Tap the right page to turn forward, the left page to turn back.
	const tap = React.useMemo(
		() =>
			Gesture.Tap()
				.runOnJS(true)
				.onEnd((e, success) => {
					if (success) e.x > spine ? ref.current?.next() : ref.current?.prev();
				}),
		[spine],
	);

	const renderItem = React.useCallback(
		({ item, relativeProgress }: CarouselRenderItemInfo<number>) => (
			<Leaf leaf={item} page={page} relativeProgress={relativeProgress} />
		),
		[pageWidth, lift],
	);

	const fill = useAnimatedStyle(() => ({
		transform: [{ scaleX: Math.max(0, Math.min(1, progress.value / LAST)) }],
	}));
	const poem = POEMS[shown - 1];
	const caption = poem?.title ?? (shown === 0 ? "Six Tang Poems" : "The End");
	const detail = poem
		? `${poem.author} · page ${shown} of ${POEMS.length}`
		: shown === 0
			? "Cover"
			: "Colophon";

	return (
		<View style={styles.fill}>
			<Svg width={width} height={height} style={StyleSheet.absoluteFill}>
				<Rect width={width} height={height} fill="#4A372B" />
				<Path
					d={repeat(
						Math.ceil(width / 7),
						(k) => `M${k * 7 + 3} 0h4v${height}h-4z`,
					)}
					fill="#503C2F"
				/>
			</Svg>
			<LinearGradient
				colors={["rgba(0,0,0,0.35)", "transparent"]}
				style={[styles.topShade, { height: insets.top + 80 }]}
			/>

			<View style={[styles.topBar, { marginTop: insets.top }]}>
				<View style={styles.barSide}>
					<Chevron />
					<Text style={styles.barText}>Library</Text>
				</View>
				<Text style={styles.bookTitle}>Six Tang Poems</Text>
				<Text style={[styles.barSide, styles.aa]}>Aa</Text>
			</View>

			<View style={styles.desk}>
				<View style={stage}>
					<View
						style={[
							styles.board,
							{
								left: spine - pageWidth - 5,
								top: lift - 6,
								width: 2 * pageWidth + 10,
								height: pageHeight + 12,
							},
						]}
					/>
					<View style={[styles.side, page, { left: spine - pageWidth }]}>
						<View
							style={[
								styles.sheet,
								{ transform: [{ scale: pageWidth / PAGE.width }] },
							]}
						>
							<Endpaper side="left">
								<Text style={[styles.exLibrisBox, styles.exLibris]}>
									EX LIBRIS
								</Text>
							</Endpaper>
						</View>
					</View>
					<GestureDetector gesture={tap}>
						<View collapsable={false} style={StyleSheet.absoluteFill}>
							<Carousel
								ref={ref}
								data={LEAVES}
								loop={false}
								autoplay={settled !== LAST}
								autoplayInterval={INTERVAL}
								itemSize={pageWidth}
								style={stage}
								animation={{
									type: "timing",
									duration: 900,
									easing: EASE_OUT_QUART,
								}}
								progress={progress}
								itemAnimation={itemAnimation}
								onSnapToItem={setSettled}
								renderItem={renderItem}
							/>
						</View>
					</GestureDetector>
					<LinearGradient
						colors={GUTTER}
						locations={[0, 0.45, 0.5, 0.55, 1]}
						start={{ x: 0, y: 0 }}
						end={{ x: 1, y: 0 }}
						style={[
							styles.gutter,
							{ left: spine - 8, top: lift, height: pageHeight },
						]}
					/>
				</View>
			</View>

			<View
				style={[
					styles.bottom,
					{ paddingBottom: Math.max(insets.bottom, 12) + 8 },
				]}
			>
				<View style={styles.track}>
					<Animated.View style={[styles.trackFill, fill]} />
				</View>
				<View style={styles.nav}>
					<Pressable
						accessibilityLabel="Previous page"
						onPress={() => ref.current?.prev()}
						style={[styles.navButton, styles.center]}
					>
						<Chevron />
					</Pressable>
					<View style={styles.captionBox}>
						<Text style={styles.caption} numberOfLines={1}>
							{caption}
						</Text>
						<Text style={styles.detail}>{detail.toUpperCase()}</Text>
					</View>
					<Pressable
						accessibilityLabel="Next page"
						onPress={() => ref.current?.next()}
						style={[styles.navButton, styles.center]}
					>
						<Chevron flip />
					</Pressable>
				</View>
			</View>
		</View>
	);
}
